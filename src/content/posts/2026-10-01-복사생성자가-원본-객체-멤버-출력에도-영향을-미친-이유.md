생성자, 복사 생성자, 소멸자가 순서대로 호출되어서 생성/소멸 안내문이 의도대로 출력되는 것을 확인할 수 있었다. 하지만 객체의 저장된 내용을 출력하는 result()에서 원본 객체와 복사된 객체 둘 다 이상한 값이 나오는 오류가 있었다.

입력값에 뭐를 넣든 60점 이상의 학생 수는 0명, 평균은 e로 지수를 나타내는 숫자 표현이 고정되어 출력되었다.

```
#include <iostream>
using namespace std;

class Dept {
	int size; // scores 배열 크기
	int* scores; // 동적할당받을 정수배열 주소
public:
	Dept(int size) { // 일반 생성자
		this->size = size;
		scores = new int[size];
		cout << "(일반 생성자 실행)" << endl;
	}
	Dept(const Dept& dept); // 복사 생성자
	~Dept(); // 소멸자

	int getSize() { return size; }
	void input();						// size만큼 반복하면서 키보드로 정수를 받아 scores 배열에 저장
	bool over60(int index);				// index 위치의 학생성적이 60이상이면 true 반환
	int getSum();						// 학생 점수의 합계 계산(반복문)
};


//***문제부분***
Dept::Dept(const Dept& dept) {			//복사 생성자 정의
	size = dept.size;					//복사된 객체의 size에 인자로 받은 객체의 size를 대입
	scores = new int[dept.size];				//복사하려는 객체의 사이즈만큼 동적 할당
	cout << "(복사 생성자 실행)" << endl;
}
//******

Dept::~Dept() {							//소멸자 정의
	delete[] scores;					//동적 할당 delete
	cout << "(소멸자 실행)" << endl;
}

void Dept::input() {					//input()정의
	for (int i = 0; i < size; i++) {	//객체의 사이즈만큼 반복해서 점수 입력받기
		cin >> scores[i];
	}
}

bool Dept::over60(int index) {			//60이상인 학생수 반환하는함수 정의
	if (scores[index] >= 60) {			//학생 점수가 60이상인 경우 true 반환
		return true;
	}
	else {
		return false;					//60미만인 경우 false 반환
	}
}

int Dept::getSum() {					//학과 학생 점수의 합 구하기
	int sum = 0;						//점수 합 초기화
	for (int i = 0; i < getSize(); i++) {	//학생 수만큼 반복
		sum += scores[i];					//i번째 학생 점수 더하기
	}

	return sum;
}




void result(Dept dept, int& cnt60, double& avg) {	//결과 출력 함수 정의
	for (int i = 0; i < dept.getSize(); i++) {		//사이즈만큼 반복
		if (dept.over60(i)) {						//i번째 값이 60이상이면 cnt60 1증가
			cnt60++;
		}
	}
	avg = (double)(dept.getSum()) / dept.getSize();		//평균 구해 avg 에 저장
}






int main() {
	cout << "<학과 1>" << endl << "학생수: ";
	int n;
	cin >> n;

	cout << n << "개 점수 입력" << endl;
	Dept dept1(n); // 객체 생성
	dept1.input(); // n개 점수 입력

	int cnt60 = 0; // 60점 이상 학생수
	double avg = 0; // 점수 평균



	result(dept1, cnt60, avg); // 결과 생성
	cout << "60 이상은 " << cnt60 << "명" << endl;
	cout << "점수 평균은 " << avg << "점" << endl << endl;
	cout << "<학과 2>" << endl;



	cnt60 = 0; // 값 초기화
	avg = 0; // 값 초기화

	Dept dept2(dept1); // 새로운 객체 생성
	result(dept2, cnt60, avg); // 결과 생성
	cout << "60 이상은 " << cnt60 << "명" << endl;
	cout << "점수 평균은 " << avg << "점" << endl;
}
```

확인 결과 복사생성자에서 scores 리스트의 복사가 완료되지 않은 것을 알 수 있었다.

```
//***문제부분***
Dept::Dept(const Dept& dept) {			//복사 생성자 정의
	size = dept.size;					//복사된 객체의 size에 인자로 받은 객체의 size를 대입
	scores = new int[dept.size];				//복사하려는 객체의 사이즈만큼 동적 할당
	for (int i = 0; i < size; i++) {			//객체의 원소 복사해오기
		scores[i] = dept.scores[i];
	}
	cout << "(복사 생성자 실행)" << endl;
}
```

복사생성자가 제대로 정의되지 않은 것이다. 동적할당 되었음에도 초기화되지 않은 메모리에는 쓰레기값이 들어가게되고, 그 결과 멤버변수 값이 이상하게 출력되었던 것이다.

그렇다면 첫 번째 객체는 복사되지 않았으므로 제대로 출력되어야 하는 것이 아닌지 의문을 갖게 되었다.

첫 번째 객체의 result() 결과도 이상하게 떴던 이유는 복사생성자의 호출 방법에 있었다.

#### 복사생성자 호출 상황

1\. 객체를 Call by Value로 함수에 넘겨줄 때

myFunc(Dept dept1) { ... }

2\. 복사 생성자 그대로 사용할 때

Dept(Dept dept1);

3\. 함수에서 객체를 값으로 반환할 때

...

return dept1;

다음과 같이 객체가 복사되는 상황에서는 복사생성자의 프로토타입을 따르지 않더라도 복사생성자가 실행된다.

즉, 첫 번째 객체의 result()를 호출했을 당시 객체를 Call by Value로 전달하면서 객체를 복사해 간 후 복사한 객체의 멤버함수를 호출한 것이기 때문에 두 번째 객체와 마찬가지로 쓰레기값이 출력되었던 것이다. 

복사 생성자를 정의하면서 나도 모르게 Dept(Dept d) 형태로 함수를 호출해야 한다고 생각했던 것이 오류의 원인이었다.
